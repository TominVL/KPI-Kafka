CREATE TABLE IF NOT EXISTS boilers_by_block (
  block_id text,              -- напр
  ts timestamp,               -- момент вимірювання
  pressure_mpa double,        -- тиск пари
  temperature_c double,       -- температура
  fuel_flow_tph double,       -- витрата палива, т/год
  fuel_type text,             -- "coal" | "gas" | "mixed"
  region text,                -- "Donetsk" | "Kharkiv" ...
  date date,                  -- календарна дата (для агрегацій/партицій)
  PRIMARY KEY ((block_id), ts)
) WITH CLUSTERING ORDER BY (ts DESC);



CREATE TABLE IF NOT EXISTS turbines_by_block (
  block_id text,
  ts timestamp,
  rpm int,                    -- оберти
  power_mw double,            -- потужність МВт
  load_pct double,            -- навантаження, %
  PRIMARY KEY ((block_id), ts)
) WITH CLUSTERING ORDER BY (ts DESC);


-- A) за block_id (добре для швидкого перегляду історії конкретного блоку)
CREATE TABLE IF NOT EXISTS emissions_by_block (
  block_id text,
  ts timestamp,
  co2_gps double,   -- грам/сек
  nox_gps double,
  so2_gps double,
  fuel_type text,
  region text,
  date date,
  PRIMARY KEY ((block_id), ts)
) WITH CLUSTERING ORDER BY (ts DESC);

-- B) за fuel_type (зручно для екологічного моніторингу по типу палива)
CREATE TABLE IF NOT EXISTS emissions_by_fuel (
  fuel_type text,   -- "coal" | "gas" | "mixed"
  ts timestamp,
  block_id text,
  co2_gps double,
  nox_gps double,
  so2_gps double,
  region text,
  date date,
  PRIMARY KEY ((fuel_type), ts, block_id)
) WITH CLUSTERING ORDER BY (ts DESC);

-- C) за (region, date) (зручно для щоденних звітів по регіону)
CREATE TABLE IF NOT EXISTS emissions_by_region_date (
  region text,
  date date,
  ts timestamp,
  block_id text,
  fuel_type text,
  co2_gps double,
  nox_gps double,
  so2_gps double,
  PRIMARY KEY ((region, date), ts, block_id)
) WITH CLUSTERING ORDER BY (ts DESC);



CREATE TABLE IF NOT EXISTS env_summary_by_station_date (
  station_id text,            -- напр. THERMAL_DN, THERMAL_KH ...
  date date,
  co2_total_t double,         -- сумарно за добу (тонни)
  nox_total_t double,
  so2_total_t double,
  avg_fuel_flow_tph double,
  PRIMARY KEY ((station_id, date))
);
